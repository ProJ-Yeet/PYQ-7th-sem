\subjecttitle{DATA MINING}{CT 72502}{3}{1}{1.5}{IV}{I}

\cobjectives
  This course introduces the fundamental principles, algorithms and applications of intelligent data processing and analysis. It will provide an in depth understanding of various concepts and popular techniques used in the field of data mining.
  \VS

\begin{enumerate}
  \item \textbf{Introduction \hfill (2 hours)}
    \begin{enumerate}
      \item Data Mining Origin
      \item Data Mining \& Data Warehousing basics
    \end{enumerate}
    \vs

  \item \textbf{Data Preprocessing \hfill (6 hours)}
    \begin{enumerate}
      \item Data Types and Attributes
      \item Data Pre-processing
      \item OLAP \& Multidimensional Data Analysis
      \item Various Similarity Measures
    \end{enumerate}
    \vs

  \item \textbf{Classification \hfill (12 hours)}
    \begin{enumerate}
      \item Basics and Algorithms
      \item Decision Tree Classifier
      \item Rule Based Classifier
      \item Nearest Neighbor Classifier
      \item Bayesian Classifier
      \item Artificial Neural Network Classifier
      \item Issues : Overfitting, Validation, Model Comparison
    \end{enumerate}
    \vs

  \item \textbf{Association Analysis \hfill (10 hours)}
    \begin{enumerate}
      \item Basics and Algorithms
      \item Frequent Itemset Pattern \& Apriori Principle
      \item FP-Growth, FP-Tree
      \item Handling Categorical Attributes
      \item Sequential, Subgraph, and Infrequent Patterns
    \end{enumerate}
    \vs

  \item \textbf{Cluster Analysis \hfill (9 hours)}
    \begin{enumerate}
      \item Basics and Algorithms
      \item K-means Clustering
      \item Hierarchical Clustering
      \item DBSCAN Clustering
      \item Issues : Evaluation, Scalability, Comparison
    \end{enumerate}
    \vs

  \item \textbf{Anomaly / Fraud Detection \hfill (3 hours)}
    \vs

  \item \textbf{Advanced Applications \hfill (3 hours)}
    \begin{enumerate}
      \item Mining Object and Multimedia
      \item Web-mining
      \item Time-series data mining
    \end{enumerate}
    \vs
\end{enumerate}
\Vs

\noindent \textbf{Practical:}
  \begin{enumerate}[label=\arabic*.]
    \item Using either MATLAB or any other DataMining tools (such as WEKA), students should practice enough on real-world data intensive problems like IRIS or Wiki dataset.
  \end{enumerate}
  \Vs

\noindent \textbf{References:}
  \begin{enumerate}[label=\arabic*.]
    \item Ning Tan, Michael Steinbach and Vipin Kumar, Introduction to Data Mining, 2005, Addison-Wesley.
    \item Jiawei Han and Micheline Kamber, Data Mining: Concepts and Techniques, 2nd Edition, 2006, Morgan Kaufmann.
  \end{enumerate}
