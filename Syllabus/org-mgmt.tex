\subjecttitle{ORGANIZATION AND MANAGEMENT}{CT 710}{3}{2}{0}{IV}{I}

\cobjectives
  \begin{itemize}
    \item To acquire knowledge in the field of organizational management and internal organization of companies required for managing an enterprise
    \item To acquire knowledge in the field of personnel management, motivation and leadership for developing managerial skills
    \item To gain knowledge for starting a small scale unit independently
    \item To gain knowledge on case study and management information system.
  \end{itemize}
  \VS

\begin{enumerate}
  \item \textbf{Introduction \hfill (20 hours)}
    \begin{enumerate}
      \item Organization \hfill (2 hours)
        \begin{enumerate}
          \item System approach applied to Organization
          \item Necessity of Organization
          \item Principles of Organization
          \item Formal and Informal Organizations
        \end{enumerate}

      \item Management \hfill (4 hours)
        \begin{enumerate}
          \item Functions of Management
          \item Levels of Management
          \item Managerial Skills
          \item Importance of Management
          \item Models of Management
        \end{enumerate}

      \item Theory of Management \hfill (6 hours)
        \begin{enumerate}
          \item Scientific Management Approach
          \item Administrative Management Approach
          \item Behavioral Management Approach
          \item Modern Management Theories
        \end{enumerate}

      \item Forms of Ownership \hfill (2 hours)
        \begin{enumerate}
          \item Single Ownership - Advantages and limitations
          \item Partnership - Types of Partners - Advantages and limitations
          \item Joint Stock Company - Formation of Joint Stock Company - Advantages and limitations
          \item Public Corporations - Advantages and limitations
        \end{enumerate}
        
      \item Organizational Structure \hfill (2 hours)
        \begin{enumerate}
          \item Line Organization - Advantages and disadvantages
          \item Functional Organization - Advantages and disadvantages
          \item Line and Staff Organization - Advantages and disadvantages
          \item Committee Organization - Advantages and disadvantages
        \end{enumerate}

      \item Purchasing and Marketing Management \hfill (4 hours)
        \begin{enumerate}
          \item Purchasing - Introduction
          \item Functions of Purchasing Department
          \item Methods of Purchasing
          \item Marketing - Introduction
          \item Functions of Marketing
          \item Advertising
        \end{enumerate}
      
    \end{enumerate}
    \vs
  
  \item \textbf{Personal Management \hfill (8 hours)}
    \begin{enumerate}
      \item Introduction
      \item Functions of Personal Management
      \item Development of Personal Policy
      \item Manpower Planning
      \item Recruitment and Selection of manpower - Scientific selection
      \item Job Analysis, Job Evaluation and Merit Rating
      \item Wages and Incentives
    \end{enumerate}

  \item \textbf{Motivation, Leadership and Entrepreneurship \hfill (10 hours)}
    \begin{enumerate}
      \item Motivation \hfill (6 hours)
        \begin{enumerate}
          \item Human needs
          \item Maslow's Hierarchy of needs
          \item Types of Motivation
          \item Attitude Motivation; Group Motivation; Executive Motivation
          \item Techniques of Motivation
          \item Motivation Theories
            \begin{enumerate}
              \item McGregor's Theory X-Y
              \item Fear and Punishment Theory
              \item Alderfer's ERG Theory
              \item MacClelland's Theory of learned needs
              \item Herzberg's Hygiene Maintenance Theory
              \item Vroom's Expectancy / Valency Theory
            \end{enumerate}
        \end{enumerate}

      \item Leadership - Introduction \hfill (2 hours)
        \begin{enumerate}
          \item Qualities of a good leader
          \item Leadership Style
          \item Blakes and Mouton's Managerial Grid
          \item Leadership Approach
          \item Leadership Theories
        \end{enumerate}
      
      \item Entrepreneurship - Introduction \hfill (2 hours)
        \begin{enumerate}
          \item Entrepreneurship Development
          \item Entrepreneurial Characteristics
          \item Need for Promotion of Entrepreneurship
          \item Steps for establishing small scale unit
        \end{enumerate}

    \end{enumerate}

  \item \textbf{Case Studies \hfill (2 hours)}
    \begin{enumerate}
      \item Introduction
      \item Obejctives of case study
      \item Phases of case study
      \item Steps of case study
      \item Types of case studies
    \end{enumerate}
  
  \item \textbf{Management Information System \hfill (5 hours)}
    \begin{enumerate}
      \item Data and Information
      \item Need, function and Importance of MIS
      \item Evolution of MIS
      \item Organizational Structure and MIS
      \item Computers and MIS
      \item Classification of Information Systems
      \item Information Support for functional areas of management
      \item Organizing Information Systems
    \end{enumerate}
\end{enumerate}
