\subjecttitle{DIGITAL SIGNAL ANALYSIS AND PROCESSING}{EX 710}{3}{1}{1.5}{IV}{I}

\cobjectives
  To introduce digital signal processing techniques and algorithms
  \VS

\begin{enumerate}
  \item \textbf{Discrete time signals and systems \hfill (8 hours)}
    \begin{enumerate}
      \item Discrete time signal, basic signal types
      \item Energy signal, power signal
      \item Periodicity of discrete time signal
      \item Transformation of independent variable
      \item Discrete time Fourier series and properties
      \item Discrete time Fourier transform and properties
      \item Discrete time system properties
      \item Linear time invariant (LTI) system convolution sum, properties of LTI system
      \item Frequency response of LTI system
      \item Sampling of continuous time signal, spectral properties of sampled signal.
    \end{enumerate}
    \vs

  \item \textbf{Z-Transform \hfill (4 hours)}
    \begin{enumerate}
      \item Defintion, convergence of Z-transform and region of convergence
      \item Properties of Z-transform (linearity, time shift, multiplication by exponential sequence, differentiation, time reversal, convolution, multiplication)
      \item Inverse z-transform by long division and partial fraction expansion.
    \end{enumerate}
    \vs

  \item \textbf{Analysis of LTI system in frequency domain \hfill (6 hours)}
    \begin{enumerate}
      \item Frequency response of LTI system, response to complex exponential
      \item Linear constant co-efficient difference equation and corresponding system function
      \item Relationship of frequency response to pole-zero of system
      \item Linear phase of LTI system and its relationship to causality.
    \end{enumerate}
    \vs

  \item \textbf{Discrete filter structures  \hfill (8 hours)}
    \begin{enumerate}
      \item FIR filter, Structures for FIR filter (direct form, cascade, frequency sampling, lattice)
      \item IIR filter, structures for IIR filter (direct form I, direct form II, cascade, lattice, lattice ladder)
      \item Quantization effect (truncation, rounding), limit cycles and scaling.
    \end{enumerate}
    \vs
    
  \item \textbf{FIR filter design \hfill (6 hours)}
    \begin{enumerate}
      \item Filter design by window method, commonly used windows (rectangular window, Hanning window, Hamming window)
      \item Filter design by Kaiser window
      \item Filter design by frequency sampling method
      \item Filter design using optimum approximation, Remez exchange algorithm.
    \end{enumerate}
    \vs
    
  \item \textbf{IIR filter design \hfill (6 hours)}
    \begin{enumerate}
      \item Antenna types
      \item Propagation characteristics of microwave antennas
      \item RF an M/W radiation, safety practices and standards
    \end{enumerate}
    \vs

  \item \textbf{Discrete Fourier transform \hfill (7 hours)}
    \begin{enumerate}
      \item Discrete Fourier transform (DFT) representation, properties of DFT (linearity, time shift, frequency shift, conjugation and conjugate symmetry, duality, convolution, multiplication), circular convolution
      \item Fast Fourier Transform (FFT) algorithm (decimation in time algorithm, decimation in frequency algorithm)
      \item Computational complexity of FFT algorithm.
    \end{enumerate}
    \vs
\end{enumerate}
\Vs

\noindent \textbf{Practical:}
  \begin{enumerate}[label=\arabic*.]
    \item Introduction to DSP tools.
    \item Signal generation and manipulation
    \item Convolution
    \item Cascade of second order systems
    \item IIR filter
    \item FIR filter
  \end{enumerate}
  \Vs
