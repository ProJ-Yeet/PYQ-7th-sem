\subjecttitle{RF AND MICROWAVE ENGINEERING}{EX 716}{3}{1}{1.5}{IV}{I}

\cobjectives
  The course deals with the basic understanding of the fundamentals of Radio Frequency (RF) and Microwave (M/W) theory and applications, design and analysis practices, and measurement techniques.
  \VS

\begin{enumerate}
  \item \textbf{Introduction \hfill (3 hours)}
    \begin{enumerate}
      \item Standard frequency bands
      \item Behaviour of circuits at conventional and RF/microwave bands
      \item Microwave applications
    \end{enumerate}
    \vs

  \item \textbf{RF and M/W Transmission Lines \hfill (6 hours)}
    \begin{enumerate}
      \item Types of transmission lines
      \item Transmission line theory
      \item Smith Chart analysis
      \item Impedance transformations and matching analysis
    \end{enumerate}
    \vs

  \item \textbf{RF an M/W Network Theory and Analysis \hfill (4 hours)}
    \begin{enumerate}
      \item Scattering matrix and its properties
      \item S-Parameter derivation and analysis
    \end{enumerate}
    \vs

  \item \textbf{RF/Microwave Components and Devices \hfill (8 hours)}
    \begin{enumerate}
      \item Coupling probes
      \item Coupling loops
      \item Waveguide
      \item Termination, E-plane Tee, H-plane Tee, Magic Tee
      \item Phase-Shifter
      \item Attenuators
      \item Directional coupler
      \item Gunn diode
      \item Microwave transistor
      \item MASER
      \item Resonator and circulators
    \end{enumerate}
    \vs

  \item \textbf{Microwave Generators \hfill (5 hours)}
    \begin{enumerate}
      \item Transit-time effect
      \item Limitations of conventional tubes
      \item Two-cavity and multi-cavity klystrons
      \item Reflex klystron
      \item TWT and magnetrons
    \end{enumerate}
    \vs
    
  \item \textbf{RF Design Practices \hfill (10 hours)}
    \begin{enumerate}
      \item RF Low pass filter
        \begin{enumerate}
          \item Insertion loss
          \item Frequency scaling
          \item Microstrip implementation
        \end{enumerate}

      \item RF Amplifier
        \begin{enumerate}
          \item Amplifier theory
          \item Design and real world consideration
        \end{enumerate}

      \item Oscillator and mixer
        \begin{enumerate}
          \item Oscillator and super mixing theory
          \item Design and real world consideration
        \end{enumerate}
    \end{enumerate}
    \vs
    
  \item \textbf{Microwave Antennas and Propagation \hfill (3 hours)}
    \begin{enumerate}
      \item Antenna types
      \item Propagation characteristics of microwave antennas
      \item RF an M/W radiation, safety practices and standards
    \end{enumerate}
    \vs

  \item \textbf{RF/Microwave Measurements \hfill (6 hours)}
    \begin{enumerate}
      \item Power measurement
      \item Calorimeter method
      \item Bolometer bridge method
      \item Thermocouples
      \item Impedance measurement
      \item RF frequency measurement and spectrum analysis
      \item Measurement of unknown loads
      \item Measurement of reflection coefficient
      \item VSWR and Noise
    \end{enumerate}
\end{enumerate}
\Vs

\noindent \textbf{Practical:}
  \begin{enumerate}[label=\arabic*.]
    \item Illustration of Smith Chart and load analysis
    \item Introduction to RF and M/W signal and circuits, measuring techniques, instrumentations, and practices
    \item Designing and analysis of simple strip-line and two-port circuits using network and spectrum analysers
    \item Software-based (ADS-like) RF signal \& circuit simulation practices
  \end{enumerate}
