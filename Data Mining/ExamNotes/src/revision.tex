\T{R.2 Must Memorize \mk{(formulas you cannot derive in the hall)}}

\sbs{%
\lead{Ch 2 --- preprocessing \& similarity}
\begin{itemize}
  \item Min-max: $v' = \frac{v-min_A}{max_A-min_A}(nmax-nmin)+nmin$
  \item Z-score: $v' = \frac{v-\bar{A}}{\sigma_A}$ \ \mk{mean, not min}
  \item Decimal scaling: $v'=v/10^{\,j}$, $\max|v'|<1$
  \item Equal-width bin: $W=(B-A)/N$ \ \mk{max minus min}
  \item Minkowski: $\big(\sum|p_k-q_k|^r\big)^{1/r}$;
        $r{=}1$ Manhattan, $r{=}2$ Euclidean, $r{\to}\infty$ supremum
  \item $SMC=\frac{q+t}{q+r+s+t}$, \ $J=\frac{q}{q+r+s}$ \ \mk{Jaccard drops $t$}
  \item $\cos(d_1,d_2)=\frac{d_1\cdot d_2}{\lVert d_1\rVert \lVert d_2\rVert}$
  \item Nominal: $d=(p-m)/p$ \quad Ordinal: $z=\frac{r-1}{M-1}$
  \item PCA variance explained $=\lambda_i/\sum\lambda_j$; \ $\sum\lambda =$ trace
  \item $n$ dims $\rightarrow$ $2^n$ cuboids
\end{itemize}
\lead{Ch 3 --- classification}
\begin{itemize}
  \item $Info(D)=-\sum p_i\log_2 p_i$
  \item $Info_A(D)=\sum\frac{|D_j|}{|D|}Info(D_j)$; \ $Gain=Info(D)-Info_A(D)$
  \item $SplitInfo_A=-\sum\frac{|D_j|}{|D|}\log_2\frac{|D_j|}{|D|}$;
        \ $GainRatio=\frac{Gain}{SplitInfo}$
  \item $Gini(D)=1-\sum p_i^2$ \ \mk{binary splits only}
  \item Bayes: $P(H|X)=\frac{P(X|H)P(H)}{P(X)}$; \ naive: $P(X|C_i)=\prod P(x_k|C_i)$
  \item Laplace correction: $\frac{n_k+1}{|C_{i,D}|+k}$
  \item Sigmoid $\sigma(z)=\frac{1}{1+e^{-z}}$, \ $\sigma'=\sigma(1-\sigma)$
  \item Rule: $Laplace=\frac{n_{correct}+1}{n_{covers}+k}$
\end{itemize}}{%
\lead{Ch 3 --- the metric table \mk{(8 papers)}}
\begin{itemize}
  \item Accuracy $=\frac{TP+TN}{P+N}$ \quad Error $=1-$accuracy
  \item Sensitivity $=$ recall $=$ TPR $=\frac{TP}{P}$
  \item Specificity $=$ TNR $=\frac{TN}{N}$
  \item FPR $=$ false alarm $=\frac{FP}{N}=1-$specificity
  \item Precision $=\frac{TP}{TP+FP}$
  \item $F_1=\frac{2\,prec\cdot rec}{prec+rec}$
  \item \mk{$P=TP+FN$ (actual positives), $N=FP+TN$}
\end{itemize}
\lead{Ch 4 --- association}
\begin{itemize}
  \item $support(A\Rightarrow B)=P(A\cup B)$
  \item $confidence(A\Rightarrow B)=\frac{sup(A\cup B)}{sup(A)}$
  \item $lift=\frac{P(A\cup B)}{P(A)P(B)}$ \ \mk{$>$1 positive, $<$1 negative}
  \item $\chi^2=\sum\frac{(obs-exp)^2}{exp}$
  \item $all\_conf=\frac{sup(A\cup B)}{\max\{sup(A),sup(B)\}}$;
        \ $Kulc=\frac{1}{2}(P(A|B)+P(B|A))$
  \item $k$-itemset $\rightarrow$ $2^k-2$ candidate rules
\end{itemize}
\lead{Ch 5 --- clustering}
\begin{itemize}
  \item $SSE=\sum_i\sum_{x\in C_i} dist(x,c_i)^2$
  \item Silhouette $s=\frac{b-a}{\max(a,b)}$
  \item DBSCAN: core if $|N_\varepsilon(p)|\ge$ MinPts
\end{itemize}
\lead{Ch 6 \& 7}
\begin{itemize}
  \item DB$(r,\pi)$-outlier: $\frac{\lVert\{o'|dist(o,o')\le r\}\rVert}{\lVert D\rVert}\le\pi$
  \item $P(I|A)=\frac{P(A|I)P(I)}{P(A|I)P(I)+P(A|\lnot I)P(\lnot I)}$
  \item $PR(A)=(1-d)+d\sum\frac{PR(T_i)}{C(T_i)}$, \ $d=0.85$
\end{itemize}}
