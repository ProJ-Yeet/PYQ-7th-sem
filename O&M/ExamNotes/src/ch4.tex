\T{4.1 Introduction and Objectives}

\Q{What is a case study? Why are case studies conducted? Objectives and value of case study}{\m{2}\m{3}\m{4}}
{\color{sub}\footnotesize \tS{8} \yr{81 Ba, 80 Ba, \textbf{73 Ch}, \textbf{71 Ch}, 71 Shr, 70 Asa, \textbf{69 Ch}, \textbf{66 Bh}} \gap$\cdot$\gap short note in 71 Shr, 69 Ch, 66 Bh (``value of case study'')}

\sbs{%
\lead{Meaning}
\begin{itemize}
  \item A \textbf{case} = written description of a \textbf{real management situation}:
    facts, reports, even opinions, but \textbf{no evaluation}.
  \item \textbf{Case study method}: learners analyse the case, identify problems, weigh
    alternatives and \textbf{decide}, as a manager would.
  \item Belief behind it: managerial competence comes from studying and discussing
    \textbf{actual situations}, not only theory.
  \item Word means different things to different users:
    \begin{itemize}
      \item lawyer: a legal case;
      \item researcher: a vehicle for \textbf{testing a hypothesis};
      \item management education: an \textbf{authentic problem} to train analysis and
        decision making.
    \end{itemize}
  \item Aim secured by facing the learner with a \textbf{whole series} of cases, not one.
  \item Harvard Business School made it the standard method of management teaching
    (1920s) \added.
\end{itemize}
\lead{Why conducted (AJ Sir)}
\begin{itemize}
  \item Teach managerial skills: \textbf{analyse} a situation, \textbf{frame} the right
    questions, \textbf{find answers}, \textbf{make decisions}.
  \item Link the three foundations of management practice: \textbf{research, theory,
    practice}.
\end{itemize}}{%
\lead{Objectives}
\begin{enumerate}
  \item \textbf{Impart knowledge and facts} about real organizations.
  \item Develop \textbf{problem analysis} skill: separate symptoms from causes.
  \item Develop \textbf{decision-making} skill under incomplete information.
  \item Improve \textbf{communication}: discuss, defend, write reports.
  \item Shape \textbf{attitude}: nothing in human behaviour is absolutely right or wrong;
    respect other views.
  \item Apply \textbf{theory to practice}; test principles against reality.
  \item Build \textbf{teamwork} and judgement through group discussion.
\end{enumerate}}

\sbs{%
\lead{Value (short note, 66 Bh)}
\begin{itemize}
  \item $+$ realistic, active learning.
  \item $+$ develops analytical thinking; learner participates.
  \item $+$ many solutions welcome $\rightarrow$ no single ``right'' answer.
  \item $+$ bridges classroom and industry.
  \item $-$ time consuming: gathering, arranging and writing up the facts is long and
    tedious.
  \item $-$ needs a skilled discussion leader.
  \item $-$ one case may not generalize.
\end{itemize}}{%
\lead{Case study vs lecture \added}
{\small
\begin{tabularx}{\linewidth}{@{}L{1.5cm}XX@{}}
\toprule
\hrow & \thead{Lecture} & \thead{Case study} \\
\midrule
Learner & listens & analyses, decides \\
Content & principles & real situation \\
Answer & one correct & several defensible \\
Skill built & recall & judgement, communication \\
\bottomrule
\end{tabularx}}}

\creamq{Types of case studies}{\m{2}}
{\color{sub}\footnotesize \tP{1} \yr{80 Ba}}
\sbs{%
\lead{By research purpose}
\begin{itemize}
  \item \textbf{Illustrative}: descriptive; one or two instances show what a situation is
    like; makes the unfamiliar familiar; gives readers a common language.
  \item \textbf{Exploratory (pilot)}: short study before a large investigation, to find the
    right questions and measures. Risk: early findings released as conclusions.
  \item \textbf{Cumulative}: combines past studies from several sites and times $\rightarrow$
    wider generalization at no new cost.
  \item \textbf{Critical instance}: one or a few sites of unique interest, or to challenge a
    universal claim; answers cause-effect questions.
\end{itemize}}{%
\lead{Teaching cases}
\begin{itemize}
  \item \textbf{Incident}: a short event to react to.
  \item \textbf{Problem / decision}: choose an action.
  \item \textbf{Appraisal}: evaluate a situation.
  \item \textbf{Descriptive}: shows how a firm handled something.
\end{itemize}}

\T{4.2 Phases and Steps of Case Study}

\Q{Explain the phases and steps of case study, with an example from your field}{\m{4}\m{5}\m{8}}
{\color{sub}\footnotesize \tF{5} \yr{\textbf{82 Bh}, \textbf{81 Bh}, 81 Ba, \textbf{80 Bh}, \textbf{79 Bh}} \gap$\cdot$\gap short note in 82 Bh, 81 Bh \gap$\cdot$\gap \mk{every paper since 2079 except 80 Ba}}

\sbs{%
\lead{Five phases (group discussion)}
\figT{c4_phases.png}}{%
\lead{Notes on the phases}
\begin{itemize}
  \item \textbf{Phase 1}: every member reads the case alone first; a group that has not
    read cannot discuss.
  \item \textbf{Phase 2}: facts are organized by relevance to the decision; irrelevant
    detail is dropped.
  \item \textbf{Phase 3}: some cases supply the critical questions; otherwise the group
    phrases them. A wrongly defined problem wastes the rest.
  \item \textbf{Phase 4}: each sub-group presents its strongest argument; decisions are
    then debated openly.
  \item \textbf{Phase 5}: the lesson is transferred beyond this one case.
  \item The \textbf{discussion leader} (teacher) guides but does not give the answer.
\end{itemize}}

\sbs{%
\lead{Six steps (individual analysis)}
\figT{c4_steps.png}}{%
\lead{Notes on the steps}
\begin{itemize}
  \item \textbf{Step 1}: read the case and its questions, then write in one line what the
    case is about and what action or answer is required.
  \item \textbf{Step 2}: arrange facts systematically; mark which are facts and which are
    opinions of people in the case.
  \item \textbf{Step 3}: treat first ideas as tentative; give every alternative due
    consideration before rejecting it.
  \item \textbf{Step 4}: choose \textbf{one} solution, the best answer on the given data.
  \item \textbf{Step 6 checklist}: what does it achieve? what difficulties? what limits
    (time, money, equipment, manpower)? is it the best alternative?
  \item A solution that fails the checklist sends you back to step 3.
\end{itemize}}

\creamq{Phases vs steps: which to write}{}
\sbs{%
\begin{itemize}
  \item \textbf{Phases} = how a \textbf{group} works through a case in a class or
    workshop (discussion method).
  \item \textbf{Steps} = how \textbf{one analyst} solves it and writes it up (report
    method).
\end{itemize}}{%
\begin{itemize}
  \item Question says ``phases'' $\rightarrow$ 5 phases; ``steps'' $\rightarrow$ 6 steps;
    ``phases and steps'' (81 Bh) $\rightarrow$ both lists, 2 marks each.
  \item Always add one line of example from engineering (below).
\end{itemize}}

\lead{Analysing a case problem (AJ Sir): the reading order before step 1}
\begin{enumerate}
  \item Read the \textbf{title} of the case.
  \item Note the \textbf{company} and the case \textbf{problems} as given.
  \item \textbf{Careful, detailed reading} of the whole case.
  \item \textbf{Meet the central characters}: who decides, who is affected.
  \item \textbf{Define the problems}, then develop the solution.
\end{enumerate}

\lead{Example from engineering: why did a hydropower project finish two years late? \added}
{\small
\begin{tabularx}{\textwidth}{@{}L{3.1cm}X@{}}
\toprule
\hrow \thead{Step} & \thead{Applied} \\
\midrule
1 State the problem & 60 MW project, planned 4 years, took 6; cost overrun 40\%. Why, and how to prevent it next time? \\
2 Collect and analyse data & Schedule vs actual (Gantt), change orders, land-acquisition dates, contractor payments, site logs, interviews with engineers and locals. Finding: land dispute (8 months), tunnel geology surprise (10 months), late payments $\rightarrow$ contractor slowdown. \\
3 Tentative solutions & (a) better geological survey; (b) settle land before tender; (c) escrow payment to contractor; (d) matrix project organization with one accountable manager. \\
4 Recommended solution & Combine (a)+(b) as pre-construction gates, (c) for cash flow; (d) as the structure. \\
5 Written report & Background, problem, data, analysis, recommendations, implementation plan. \\
6 Checklist & Achieves on-time completion; needs upfront survey budget; limit: political land issues; still the best alternative. \\
\bottomrule
\end{tabularx}}

\T{4.3 Case Report Writing, and Preparing a Case Study}

\Q{Prepare a case study of an organization (NEA / Nepal Telecom; your college) following the case structure}{\m{8}}
{\color{sub}\footnotesize \tP{2} \yr{\textbf{74 Ch}, \textbf{72 Ch}} \gap$\cdot$\gap applied 8-markers: fill in the structure \gap$\cdot$\gap full models in $\S$4.4 (72 Ch) and $\S$4.5 (74 Ch)}

\sbs{%
\lead{Case report writing methodology}
\begin{enumerate}
  \item \textbf{Organizational background}:
    \begin{itemize}
      \item goals and objectives; formation;
      \item structure; social needs and value;
      \item historical relation; financial position.
    \end{itemize}
  \item \textbf{Identification of the problem}.
  \item \textbf{Hypothesis}: the suspected cause.
  \item \textbf{Case research design}: what data, from whom, how.
  \item \textbf{Collection and analysis of data}.
  \item \textbf{Generalization and interpretation}.
  \item \textbf{Conclusions and recommendations}.
\end{enumerate}}{%
\lead{Typical report titles (a full O\&M case study)}
\begin{enumerate}
  \item Acknowledgement.
  \item Organization: history, objectives, input, process, output.
  \item Organization structure (chart).
  \item Form of ownership.
  \item Personnel policies; manpower planning.
  \item Recruitment and selection; training and development.
  \item Job evaluation; merit rating; wages and incentives.
  \item Motivation; marketing; leadership; MIS.
  \item Drawbacks; suggestions and recommendations.
  \item Conclusion; references.
\end{enumerate}
\begin{itemize}
  \item The titles are the syllabus itself: a case study is where Ch.\,1--3 and 5 are
    applied to one real organization.
\end{itemize}}

\lead{Writing the 8-mark version in the exam}
\sbs{%
\begin{itemize}
  \item Use the \textbf{7 headings} of the methodology as your headings; 1--3 lines each.
  \item Spend the marks on \textbf{what the question names}: 72 Ch names planning horizon,
    leadership, motivation, HRD $\rightarrow$ one analysis paragraph each.
  \item Give \textbf{one or two real numbers} (loss \%, load-shedding hours, pass rate);
    they show the data step was done.
\end{itemize}}{%
\begin{itemize}
  \item Draw \textbf{one} figure: an org chart, a cause-and-effect (fishbone) diagram, or a
    time-bound plan.
  \item End with \textbf{recommendations + time frame}, then the \textbf{checklist} line
    (achieves, difficulties, limits, best alternative).
  \item Roughly: background 1, problem + hypothesis 1, design + data 2, analysis 2,
    recommendations 2 marks.
\end{itemize}}

\T{4.4 Mock Case Study 1: Nepal Electricity Authority (72 Ch)}

{\color{sub}\footnotesize \tP{1} \yr{\textbf{72 Ch}} \gap$\cdot$\gap written out in full as a model \gap$\cdot$\gap figures are NEA's published figures; staff-survey results are illustrative \gap \added}

\lead{Question (72 Ch, 8 marks)}
{\small\itshape If you are asked to prepare the case study considering the planning horizon,
leadership, motivation and human resource development for either Nepal Electricity Authority
or Nepal Telecom to improve the existing performance of these institutions, how do you prepare
the case study following its structure?}

\lead{Title}
\textbf{Improving the performance of Nepal Electricity Authority: a case study of planning
horizon, leadership, motivation and human resource development.} Period studied: FY 2072/73
(the crisis year) to FY 2080/81.

\lead{Acknowledgement}
Thanks to NEA's Planning, Monitoring and IT Directorate for the annual reports, to the
distribution-centre chiefs who gave interviews, and to the course teacher for guidance.

\sbs{%
\lead{1. Organizational background}
\begin{itemize}
  \item \textbf{Formation}: established 16 August 1985 (Bhadra 1, 2042) under the
    \textbf{NEA Act 2041}, merging the Electricity Department, Nepal Electricity
    Corporation and related development boards.
  \item \textbf{Form of ownership}: \textbf{public corporation}, fully government owned
    ($\S$1.4): own legal entity, govt control, service + commercial motive.
  \item \textbf{Goals}: generate, transmit and distribute adequate, reliable and affordable
    power; plan and develop the national grid; trade power with India.
  \item \textbf{Structure}: Board of Directors chaired by the Energy Minister $\rightarrow$
    Managing Director $\rightarrow$ Deputy Managing Directors heading functional
    directorates (figure) $\rightarrow$ regional offices and distribution centres.
    A \textbf{line and staff} structure ($\S$1.5).
  \item \textbf{Social value}: power drives industry, health, education, irrigation,
    telecom; every household is a customer.
  \item \textbf{Financial position}: accumulated loss of \textbf{Rs 34.61 bn} up to FY
    2072/73; first net profit \textbf{Rs 1.46 bn} in FY 2073/74, profitable every year since.
\end{itemize}}{%
\lead{Input, process, output}
{\small
\begin{tabularx}{\linewidth}{@{}L{1.25cm}X@{}}
\toprule
\hrow \thead{Stage} & \thead{NEA} \\
\midrule
Input & water (river flow), own plants, power bought from IPPs, imports from India, staff, capital, tariff revenue \\
Process & generation $\rightarrow$ transmission (132/220/400 kV) $\rightarrow$ distribution $\rightarrow$ metering and billing \\
Output & electricity to domestic, industrial, commercial consumers; export in the wet season; revenue \\
\bottomrule
\end{tabularx}}}

\figC{c4_nea_org.png}{\textwidth}
{\small\color{sub}\centering NEA organization structure (simplified to directorate level).\par}

\sbs{%
\lead{2. Identification of the problem}
\begin{itemize}
  \item In FY 2072/73 NEA showed every sign of a failing public enterprise:
    \begin{itemize}
      \item \textbf{load-shedding} up to \textbf{18 hours a day} in the dry winter;
      \item \textbf{system loss 25.78\%}: a quarter of energy lost, much of it theft;
      \item accumulated \textbf{loss Rs 34.61 bn}; industries on diesel generators;
      \item low staff morale; public anger.
    \end{itemize}
  \item \textbf{Problem statement}: why did NEA perform so poorly, and what should it do in
    planning, leadership, motivation and HRD to improve?
\end{itemize}
\lead{3. Hypothesis}
\begin{itemize}
  \item \textbf{Main}: poor performance is as much a \textbf{management} failure as a
    shortage of generation.
  \item H1: planning horizon was short $\rightarrow$ fire-fighting, no storage.
  \item H2: leadership was weak and unstable $\rightarrow$ no accountability.
  \item H3: staff lacked motivators $\rightarrow$ leakage tolerated.
  \item H4: HRD was neglected $\rightarrow$ skill gaps in new technology.
\end{itemize}}{%
\lead{4. Case research design}
{\small
\begin{tabularx}{\linewidth}{@{}L{1.45cm}X@{}}
\toprule
\hrow \thead{Data} & \thead{Source and method} \\
\midrule
Secondary & NEA annual reports FY 2072/73--2080/81: loss, demand, supply, finance; news reports \\
Interviews & 20 officers: MD office, 2 directorates, 6 distribution centres \\
Staff survey & 200 staff, 5-point scale: pay, recognition, training, leadership \\
Consumer survey & 300 households and 30 industries: supply hours, complaints \\
Observation & 2 substations and 1 control centre (dispatch) \\
Analysis & trend charts; before/after 2073; fishbone of causes \\
\bottomrule
\end{tabularx}}
\lead{5. Collection and analysis of data}
\begin{itemize}
  \item \textbf{System loss}, FY 72/73 to 80/81 (\%): \textbf{25.78} $\rightarrow$ 22.90
    $\rightarrow$ 20.45 $\rightarrow$ 15.32 $\rightarrow$ 15.27 $\rightarrow$ 17.18 (COVID
    year) $\rightarrow$ 15.38 $\rightarrow$ 13.46 $\rightarrow$ \textbf{12.73}.
  \item The sharpest fall came in the first three years after the new MD (2073 Bh).
  \item Each point of loss is roughly a billion rupees a year: NEA says leakage control
    earns it about Rs 12 bn a year.
\end{itemize}}

\figC{c4_nea_fishbone.png}{0.8\textwidth}
{\small\color{sub}\centering Cause-and-effect analysis of NEA's performance in FY 2072/73.\par}

\sbs{%
\lead{(a) Planning horizon}
\begin{itemize}
  \item Found: planning mostly \textbf{short term}, one dry season at a time.
  \item Storage and peaking plants delayed for years $\rightarrow$ winter deficit grew with
    demand.
  \item Fix by three horizons:
    \begin{itemize}
      \item \textbf{short} (1 yr): manage peak, cut leakage, import in winter;
      \item \textbf{medium} (2--5 yr): transmission lines and substations, smart meters;
      \item \textbf{long} (10--20 yr): storage projects, cross-border lines, export.
    \end{itemize}
  \item H1 \textbf{accepted}.
\end{itemize}
\lead{(b) Leadership}
\begin{itemize}
  \item A new MD took charge on 14 Sept 2016 (Bhadra 2073).
  \item Within about two months Kathmandu Valley was declared load-shedding free (Tihar,
    Oct 2016); nationwide by \textbf{May 2018}.
  \item Mostly the \textbf{same} generation capacity: the change came from
    \textbf{dispatch management}, ending \textbf{dedicated-feeder} privileges, leakage raids,
    peak-hour imports.
  \item Style: high concern for task \textbf{and} people $\rightarrow$ 9,9 team management
    on Blake--Mouton's grid ($\S$3.3); visible, decisive, accountable.
  \item H2 \textbf{accepted}: leadership was the turning factor.
\end{itemize}}{%
\lead{(c) Motivation}
\begin{itemize}
  \item Staff survey (illustrative): pay and job security rated fair; recognition and
    growth rated poor.
  \item Herzberg ($\S$3.2): \textbf{hygiene} factors present, \textbf{motivators} absent
    $\rightarrow$ no dissatisfaction, but no drive either.
  \item What worked: loss \textbf{targets per distribution centre}, monthly ranking,
    public recognition, action against poor performers.
  \item Maslow: esteem needs met by recognition; Vroom: effort $\rightarrow$ visible result
    $\rightarrow$ reward raised expectancy.
  \item H3 \textbf{accepted}.
\end{itemize}
\lead{(d) Human resource development}
\begin{itemize}
  \item Found: little structured training; promotion by seniority; no succession plan.
  \item New needs: smart metering, SCADA, grid operation, cross-border trade, project
    management.
  \item Needed: training-needs analysis ($\S$2.4), skill-based training, merit rating
    linked to promotion, succession planning for key posts.
  \item H4 \textbf{partly accepted}: training exists but is not tied to career growth.
\end{itemize}}

\sbs{%
\lead{6. Generalization and interpretation}
\begin{itemize}
  \item The turnaround came \textbf{before} big new plants: management, not only megawatts,
    explains most of it $\rightarrow$ main hypothesis supported.
  \item Supply growth (new plants, imports) then made the gains last.
  \item \textbf{Generalization}: other Nepali public enterprises (Nepal Telecom, Nepal
    Airlines, Nepal Oil Corporation) can improve the same way: stable accountable
    leadership, performance targets, a long planning horizon, merit-based HRD.
  \item \textbf{Limit}: the gains depend on one leader unless built into systems;
    political interference is the main risk.
\end{itemize}}{%
\lead{7. Conclusions and recommendations}
{\small
\begin{tabularx}{\linewidth}{@{}L{1.35cm}XL{1.1cm}@{}}
\toprule
\hrow \thead{Area} & \thead{Recommendation} & \thead{Horizon} \\
\midrule
Planning & rolling 1/5/20-year plan; load forecast; storage first & all three \\
Leadership & fixed MD tenure; performance contract with the Board & short \\
Motivation & loss and service targets per centre; ranking, rewards & short \\
HRD & needs analysis; training academy; merit promotion; succession & medium \\
Technology & smart meters, SCADA, MIS dashboards ($\S$5) & medium \\
Governance & shield operations from political transfers & long \\
\bottomrule
\end{tabularx}}}

\sbs{%
\lead{Checklist (step 6)}
\begin{itemize}
  \item \textbf{Achieves}: reliable supply, loss near 10\%, sustained profit, motivated staff.
  \item \textbf{Difficulties}: union resistance to performance ranking; political transfers.
  \item \textbf{Limits}: capital for storage plants; dry-season hydrology; import capacity.
  \item \textbf{Best alternative?} Yes: cheaper and faster than building capacity alone.
\end{itemize}}{%
\lead{Drawbacks of this study; references}
\begin{itemize}
  \item Survey figures are illustrative; a real study would sample properly.
  \item References: NEA annual reports; Kathmandu Post (2016, 2017, 2019); Khabarhub (2024).
\end{itemize}}

\T{4.5 Mock Case Study 2: Recommendations to Your College (74 Ch)}

{\color{sub}\footnotesize \tP{1} \yr{\textbf{74 Ch}} \gap$\cdot$\gap written out in full as a model \gap$\cdot$\gap ``XYZ Engineering Campus'' and all its numbers are illustrative; put your own campus's facts in \gap \added}

\lead{Question (74 Ch, 8 marks)}
{\small\itshape Having spent more than 3 years in a particular college/campus of yours, what are

\sbs{%
\lead{1. Organizational background}

  \item \textbf{Programmes}: BCT, BEI, BEL, BCE; about 960 students, 72 faculty

  \item \textbf{Structure}: Board $\rightarrow$ Principal $\rightarrow$ Heads of

  \item \textbf{Financial position}: fee-funded; stable, little surplus for labs.

\lead{3. Hypothesis}

\bottomrule

    \item Middling: library 2.9, hostel 2.6.
    \item Strongest: teaching 3.4.
  \end{itemize}
  \item Faculty turnover 25\% a year; 45\% of classes taken by part-timers.
  \item Students with over 80\% attendance passed at 81\%; the rest at 44\%.
\end{itemize}}

  \item Facilities (labs, internet) hurt practical courses and the college's image.

  \item Holds for most private affiliated colleges: \textbf{people and systems} before

\lead{Abstract}
{\small This report studies ABC Cement Industries Ltd., an integrated cement plant with a
1,200 TPD kiln. It describes the company's ownership, structure, personnel practices,
motivation, leadership, marketing and MIS. The case problem is low capacity use: 64--71\%
over four years. Kiln logs for FY 2082/83 show 1,490 hours of unplanned stops, 69.1\% of them
from power trips and breakdowns. The report recommends a standby power supply, preventive
maintenance, spares planning, training and regularizing contract operators, and an ERP,
aiming at 80\% capacity use in two years.\par}}

\lead{5.2 Analysis ($\S$3.2)}
\begin{itemize}
  \item \textbf{Herzberg}:
    \begin{itemize}
      \item permanent staff: hygiene factors met (security 4.1, pay 3.4), motivators weak
        (recognition 2.4, growth 2.7) $\rightarrow$ not dissatisfied, but not driven;
      \item contract staff: even hygiene factors unmet (security 1.9) $\rightarrow$
        dissatisfaction $\rightarrow$ 28\% turnover.
    \end{itemize}
  \item \textbf{Maslow}: contract workers stuck at safety needs; permanent staff seek esteem.
  \item \textbf{Vroom}: incentive earned in 3 months of 12 $\rightarrow$ effort not seen to
    bring reward $\rightarrow$ low expectancy.
\end{itemize}
\lead{5.3 Leadership ($\S$3.3)}
\begin{itemize}
  \item \textbf{MD}: sets targets, decides fast; autocratic on purchase and hiring,
    participative in the monthly production review.
  \item \textbf{Shift engineers}: task-oriented, 9,1 on Blake--Mouton's grid: output first,
    little coaching.
  \item \textbf{Plant Manager}: visits every shift, listens; rated highest by staff
    $\rightarrow$ closest to 9,9 team management.
\end{itemize}}

\rch{Chapter 8. The Case Problem: Low Capacity Use}

\sbs{%
\lead{8.1 Identification of the problem}
\begin{itemize}
  \item Capacity used 64--71\% for four years (section 2.3), against the 85\% the budget
    assumes.
  \item Every idle kiln hour loses 50 t of clinker (1,200 TPD $\div$ 24): the loss cannot be
    made up later.
  \item \textbf{Problem statement}: why does the plant use only about two-thirds of its
    capacity, and what should management do?
\end{itemize}
\lead{8.2 Hypotheses}
\begin{itemize}
  \item \textbf{H1}: unplanned kiln stops from power trips and breakdowns are the main cause.
  \item \textbf{H2}: weak off-season demand adds to it.
  \item \textbf{H3}: manpower practices (contract turnover, little training) cause many of
    the breakdowns.
  \item \textbf{H4}: raw-material shortage is the main cause.
\end{itemize}}{%

\sbs{%
\lead{8.4 Analysis}
\begin{itemize}
  \item Kiln ran 8,760 $-$ 1,850 = 6,910 h $\rightarrow$ run factor 78.9\%.
  \item Clinker lost to unplanned stops: 1,490 h $\times$ 50 t = 74,500 t = 18.8\% of the
    396,000 t capacity.
  \item Power + breakdowns = 1,030 h = \textbf{69.1\%} of unplanned stops.
  \item \textbf{Power}: 146 trips in the year.
    \begin{itemize}
      \item each trip $\rightarrow$ about 4 h to restart and stabilize the kiln;
      \item a 5-minute trip costs hours of output.
    \end{itemize}
  \item \textbf{Breakdowns}: 29 in the maintenance log.
    \begin{itemize}
      \item 11 traced to operating error (missed lubrication, overloading);
      \item 9 of those 11 on shifts run mainly by contract operators;
      \item 6 waited for imported spares $\rightarrow$ 180 of the 420 h.
    \end{itemize}
  \item \textbf{Market}: yard-full stops all fell in monsoon and festival months.
\end{itemize}}{%

\rch{Chapter 9. Findings, Generalization and Interpretation}

\sbs{%
\lead{9.1 Strengths}
\begin{itemize}
  \item Integrated plant with own quarry $\rightarrow$ low raw-material cost.
  \item NS-marked product; brand known in 30 districts.
  \item Profitable; regular 10\% dividend.
  \item Good pay and welfare for permanent staff; lawful HR bylaws.
  \item Modern kiln with DCS control.
\end{itemize}
\lead{9.3 Generalization and interpretation}
\begin{itemize}
  \item Capacity is lost mainly \textbf{inside the plant gate} (69.1\% of unplanned stops),
    not in the market $\rightarrow$ a \textbf{management} problem.
  \item Holds for continuous-process plants (cement, steel, paper): a lost kiln hour is never
    recovered $\rightarrow$ reliability of power, maintenance and people decides output.
  \item Limit: one plant, one year of logs.
\end{itemize}}{%
\lead{9.2 Drawbacks (problems found)}
\begin{enumerate}
  \item Breakdown maintenance instead of preventive.
  \item No standby power for kiln drives.
  \item Spares bought after failure, not stocked.
  \item Two-tier workforce: contract operators insecure, untrained, 28\% turnover.
  \item Promotion by seniority; merit rating used for increments only.
  \item Production incentive too hard to earn (3 of 12 months).
  \item MIS not integrated; no real-time KPI.
  \item No off-season marketing plan.
\end{enumerate}}

\lead{Expected result}
