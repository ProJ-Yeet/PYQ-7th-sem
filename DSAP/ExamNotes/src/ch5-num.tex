\lead{The symbols, once, so nothing below is a bare number}
\begin{itemize}
  \item $\omega_p,\omega_s$ \gap pass-band and stop-band edge, radians per sample.
  \item $\Delta\omega=\omega_s-\omega_p$ \gap transition width. \mk{This, not the
        attenuation, sets the length.}
  \item $A_s$ \gap required stop-band attenuation, dB. \mk{This, not the width, sets
        the window.}
  \item $c$ \gap the chosen window's transition constant from the table, so that its
        transition width is $c\pi/M$.
  \item $M$ \gap filter order; $N=M+1$ \gap number of taps; $\alpha=(N-1)/2$ \gap group
        delay in samples.
  \item $\omega_c$ \gap cut-off of the ideal prototype; $h_d[n]$ its impulse response;
        $w[n]$ the window; $h[n]=h_d[n]\,w[n]$ the filter.
\end{itemize}

\lead{The recipe, used in \S1, \S2 and \S3}
\begin{enumerate}
  \item \textbf{Convert the spec} to $\omega$ in radians per sample. From Hz,
        $\omega=2\pi f/f_s$; from rad/s, $\omega=\Omega/f_s$; from a ripple,
        $A_s=-20\log_{10}\delta_s$.
  \item \textbf{Window} from $A_s$, taking the cheapest row of the table that clears it.
  \item \textbf{Length} $M=c\pi/\Delta\omega$, rounded \emph{up}; then $N=M+1$, rounded
        up again to the next \textbf{odd} value.
  \item \textbf{Cut-off} $\omega_c=\dfrac{\omega_p+\omega_s}{2}$, the middle of the
        transition band.
  \item \textbf{Coefficients} $h[n]=h_d[n]\,w[n]$, with
        $h_d[n]=\dfrac{\sin\big(\omega_c(n-\alpha)\big)}{\pi(n-\alpha)}$ and the centre
        tap $h_d[\alpha]=\omega_c/\pi$ done separately, because the general formula is
        $0/0$ there.
\end{enumerate}

\lead{Checks that cost nothing}
\begin{enumerate}
  \item The centre tap is always $\omega_c/\pi$ exactly. \mk{If it is not, $\omega_c$ is
        wrong.}
  \item $h[n]$ must be \textbf{symmetric} about $\alpha$: compute one half and mirror it.
  \item A \textbf{Hann}-windowed filter has $h[0]=h[N-1]=0$ exactly, because the window
        is zero there. So "the first three coefficients" begins with a zero.
  \item $N$ odd, always, so $\alpha$ is an integer.
\end{enumerate}}

\begin{asked}{2079 Chaitra \textperiodcentered{} 2075 Bhadra \textperiodcentered{} [8] [6]}
Find out the first three coefficients of impulse response of a low pass FIR filter having
Pass band edge frequency $\omega_p=0.24\pi$, Stop band edge frequency
$\omega_s=0.54\pi$ and Stop band attenuation $\alpha_s=42.2$ dB using any suitable window
function.
\end{asked}

\lead{Step 1: the window, from the attenuation alone}
\begin{itemize}
  \item Required $A_s=42.2$ dB. Read \emph{down} the table to the first row that clears
        it: Bartlett gives $25$ dB and fails, \textbf{Hann gives 44 dB and clears}.
        \mk{So Hann}, and Hamming's 53 dB would be over-design.
  \item[] \[ c=6.2 \quad\text{(Hann)} \]
\end{itemize}

\lead{Step 2: the length, from the transition width alone}
\begin{itemize}
  \item[] \[ \Delta\omega=\omega_s-\omega_p=0.54\pi-0.24\pi=0.30\pi \]
  \item[] \[ M=\frac{c\pi}{\Delta\omega}=\frac{6.2\pi}{0.30\pi}=20.67
             \ \Rightarrow\ M=21 \]
  \item[] \[ N=M+1=22\ \Rightarrow\ \boxed{N=23}\ \text{(next odd)},\qquad
             \alpha=\frac{N-1}{2}=\mathbf{11} \]
  \item \mk{Two roundings, not one.} $M$ rounds up, then $N$ rounds up again to odd.
        Stopping at $N=22$ is the commonest slip in this chapter.
\end{itemize}

\lead{Step 3: the cut-off, in the middle}
\begin{itemize}
  \item[] \[ \omega_c=\frac{\omega_p+\omega_s}{2}=\frac{0.24\pi+0.54\pi}{2}
             =\mathbf{0.39\pi}=1.2252\ \text{rad/sample} \]
  \item[] \[ h_d[n]=\frac{\sin\big(\omega_c(n-\alpha)\big)}{\pi(n-\alpha)}
             =\frac{\sin\big(0.39\pi(n-11)\big)}{\pi(n-11)},\qquad
             h_d[11]=\frac{\omega_c}{\pi}=\mathbf{0.39} \]
\end{itemize}

\lead{Step 4: the first three coefficients}
\begin{itemize}
  \item Hann window, general form then this filter's:
  \item[] \[ w[n]=0.5-0.5\cos\frac{2\pi n}{N-1}=0.5-0.5\cos\frac{2\pi n}{22} \]
  \item[] \[ w[0]=0,\qquad w[1]=0.020254,\qquad w[2]=0.079373 \]
  \item[] \[ h_d[0]=0.022865,\qquad h_d[1]=-0.009836,\qquad h_d[2]=-0.035350 \]
  \item[] \[ h[n]=h_d[n]\,w[n]\ \Rightarrow\
             \boxed{h[0]=0,\quad h[1]=-0.000199,\quad h[2]=-0.002806} \]
  \item \mk{$h[0]=0$ is not a mistake}: the Hann window is exactly zero at both ends, so
        the first and last taps of the filter vanish. Say so, because it looks like an
        error otherwise.
  \item The rest follows by symmetry, $h[22-n]=h[n]$, with the centre tap $h[11]=0.39$.
\end{itemize}

\begin{asked}{2082 Baishakh \textperiodcentered{} [2+6]}
Differentiate between analog and digital filter. Design an FIR lowpass digital filter
that will have a $-3$dB cut-off at $30\pi$ rad/s and an attenuation of $50$ dB at $45\pi$

        Hamming's $53$ dB clears, so \textbf{Hamming} and $c=6.6$:

             M=\frac{6.6\pi}{0.15\pi}=44\ \Rightarrow\ M=44,\quad N=M+1=\boxed{45},\quad
             \alpha=22 \]
  \item[] \[ \omega_c=\frac{0.3\pi+0.45\pi}{2}=\mathbf{0.375\pi},\qquad h[22]=0.375 \]

\lead{The Kaiser formulas, symbols first}
\begin{itemize}
  \item[] \[ \delta=\min(\delta_p,\delta_s),\qquad A=-20\log_{10}\delta \]
  \item[] \[ \beta=\begin{cases}
                0, & A\le21\\
                0.5842(A-21)^{0.4}+0.07886(A-21), & 21<A\le50\\
                0.1102(A-8.7), & A>50
             \end{cases} \]
  \item[] \[ M=\frac{A-8}{2.285\,\Delta\omega},\quad N=M+1\ \text{rounded up odd},\qquad
             \omega_c=\frac{\omega_p+\omega_s}{2} \]
  \item[] \[ w[n]=\frac{I_0\!\left(\beta\sqrt{1-\left(\frac{2n}{M}-1\right)^{2}}\right)}
                       {I_0(\beta)},\qquad h[n]=h_d[n]\,w[n] \]

        $2.285$ is a dimensional constant; feeding it $0.05$ instead of $0.05\pi$ is the
        most common wrong answer in this section.
\end{itemize}

        It is the single easiest mark to lose in this section, because $0.25\pi$ and
        $0.35\pi$ are the first two numbers printed.
  \item \mk{Step that is missed \#2: the ideal response is the band-pass one}, not the
        low-pass one, and each edge again sits in the middle of its own transition:
  \item[] \[ \omega_1=\frac{0.25\pi+0.35\pi}{2}=\mathbf{0.3\pi},\qquad
             \omega_2=\frac{0.6\pi+0.65\pi}{2}=\mathbf{0.625\pi} \]
  \item[] \[ h_d[n]=\frac{\sin\big(\omega_2(n-\alpha)\big)-\sin\big(\omega_1(n-\alpha)\big)}
                        {\pi(n-\alpha)},\qquad
             h_d[\alpha]=\frac{\omega_2-\omega_1}{\pi} \]
  \item The window is the same Kaiser as \S3; only $h_d$ changed.
\end{itemize}

\lead{3.4 \quad 73 Ch prints a pass band that swallows its own stop band}
{\color{sub}\footnotesize \yr{\textbf{73 Ch} \m{8}} \gap$\cdot$\gap \mk{no filter exists
as printed}}

\begin{asked}{2073 Chaitra \textperiodcentered{} [8]}
Design an FIR linear phase filter using Kaiser window to meet the following
specifications:\newline
$0.98\le|H(e^{jw})|\le1.02$, \quad for $0\ge w\ge0.9\pi$\newline
$|H(e^{jw})|\le0.01$, \quad for $0.14\pi\le w\le\pi$
\end{asked}

\begin{itemize}
  \item \mk{The step that is missed: read the spec before designing.} It gives
        $\omega_p=0.9\pi$ and $\omega_s=0.14\pi$, so $\omega_p>\omega_s$ and the two
        bands overlap. $\Delta\omega$ would be negative and $M$ negative with it.
  \item[] \[ \Delta\omega=\omega_s-\omega_p=0.14\pi-0.9\pi=-0.76\pi<0
             \ \Rightarrow\ \textbf{impossible} \]
  \item \textbf{Read $0.9\pi$ as $\mathbf{0.09\pi}$}, a dropped zero. The transition
        $0.09\pi$ to $0.14\pi$ is then sensible. Note the misprint in one line, then:
  \item[] \[ \delta_p=1.02-1=0.02,\qquad \delta_s=0.01,\qquad
             \delta=\min(0.02,0.01)=0.01 \]
  \item[] \[ A=40\ \text{dB},\qquad \beta=3.3953 \]
  \item[] \[ \Delta\omega=0.14\pi-0.09\pi=0.05\pi=0.15708,\qquad
             M=\frac{32}{2.285(0.15708)}=89.15\ \Rightarrow\ M=90,\ \boxed{N=91} \]
  \item[] \[ \omega_c=\frac{0.09\pi+0.14\pi}{2}=\mathbf{0.115\pi},\qquad \alpha=45,
             \qquad h[45]=0.115 \]
\end{itemize}

\lead{3.5 \quad 78 Ch is the only paper whose $\delta$ is not $0.01$ or $0.035$}
{\color{sub}\footnotesize \yr{\textbf{\texttt{78 Ch}} \m{8}} \gap$\cdot$\gap \mk{so it is
the only one with a different $\beta$}}

\begin{asked}{2078 Chaitra \textperiodcentered{} [8]}
Design a FIR linear phase filter using Kaiser window to meet the following
specifications:\newline
$0.98\le|H(e^{jw})|\le1.02$; \quad $0\le\omega\le0.19\pi$\newline
$|H(e^{jw})|\le0.02$; \quad $0.21\pi\le\omega\le\pi$
\end{asked}

\begin{itemize}
  \item \mk{Both ripples are $0.02$ here}, so the $\min$ does not rescue you into the
        familiar $\beta=3.3953$. Recompute it:
  \item[] \[ \delta_p=1.02-1=0.02,\qquad \delta_s=0.02,\qquad \delta=0.02 \]
  \item[] \[ A=-20\log_{10}(0.02)=\mathbf{33.98\ \text{dB}} \]
  \item[] \[ \beta=0.5842(33.98-21)^{0.4}+0.07886(33.98-21)
             =0.5842(12.98)^{0.4}+0.07886(12.98)=\mathbf{2.6523} \]
  \item[] \[ \Delta\omega=0.21\pi-0.19\pi=0.02\pi=0.062832 \]
  \item[] \[ M=\frac{A-8}{2.285\,\Delta\omega}=\frac{33.98-8}{2.285(0.062832)}
             =\frac{25.98}{0.14357}=180.95\ \Rightarrow\ M=181,\quad
             N=182\ \Rightarrow\ \boxed{N=183} \]
  \item[] \[ \omega_c=\frac{0.19\pi+0.21\pi}{2}=\mathbf{0.2\pi},\qquad \alpha=91 \]
  \item \mk{Looser ripple gives a SHORTER filter}, 183 against 75 Ash's 225 on the same
        transition width. That one-line comparison is often the last mark.
\end{itemize}

\lead{3.6 \quad 66 Ma, the only Kaiser paper worked end to end, with its own Bessel table}
{\color{sub}\footnotesize \yr{\textbf{66 Ma}} \gap$\cdot$\gap five parts, a $9$-tap
filter, and \mk{a table that gives the answer away}}

\begin{asked}{2066 Magh \textperiodcentered{} [parts a--e]}
Design a digital low-pass FIR filter with the following specifications using Kaiser

($\omega_s$) $=0.65\pi$, Pass band ripple ($\delta_p$) $=0.035$, Stop band ripple
($\delta_s$) $=0.035$.\newline
a) Find the order of filter (N) \quad b) Find the cutoff frequency ($\omega_c$)\newline
c) Find the value of shape parameter ($\beta$) \quad d) Find Kaiser window (w(n))\newline
e) Find the filter impulse response (h(n))\newline

\T{4 \quad The bookwork, which is a third of the chapter}
{\color{sub}\footnotesize \tS{31} \gap$\cdot$\gap Gibbs \tS{8} and Remez \tS{23}
\gap$\cdot$\gap \mk{175 marks of Remez alone, and not one calculation}}

\begin{itemize}
  \item \textbf{Gibbs} is asked in eight papers for 3 to 6 marks. Three sentences:
        truncating with a rectangular window convolves the ideal response with a
        kernel whose sidelobes are only $13$ dB down; the result overshoots the band
        edge by about \textbf{9 \%} with ringing either side; and \mk{lengthening the
        filter does not reduce the overshoot}, only narrows the ripples, so the cure is
        a tapered window.
  \item \textbf{Remez} is asked in twenty-three, for 4 to 10 marks, and almost always
        wants \textbf{the flowchart}. Learn the seven boxes in \S5.5 of the theory:
        input, initial guess of $r+2$ extremal frequencies, solve for $\delta$,
        interpolate and evaluate $E(\omega)$ on a dense grid, exchange the extrema,
        test against the alternation theorem, output.
  \item The three "why is it better" marks: \mk{it wastes no tolerance}, so it gives a
        \textbf{shorter filter} for the same spec; it allows \textbf{different passband
        and stopband ripple} through the weight $W(\omega)$; and it ignores the
        transition band instead of constraining it.
  \item \mk{Write $\min_h\max_\omega|W(\omega)[H_d(\omega)-H(\omega)]|$ once} and the
        "mathematical expression" marks are collected.
\end{itemize}
