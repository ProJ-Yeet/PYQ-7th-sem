\T{C. Never given \mk{(memorise all of these)}}

\lead{Amplifier: what the sheet leaves out \yr{(Ch 6, 22/24 papers)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item \textbf{Verdict}: unconditionally stable iff $K > 1$ \textbf{and} $|\Delta| < 1$,
    or equivalently $\mu > 1$. If $K < 1$ say so; do not quote $G_{T,max}$.
  \item \textbf{Root sign} in $\Gamma_S$, $\Gamma_L$: \textbf{minus when $B > 0$}, plus when
    $B < 0$, so that $|\Gamma| < 1$.
  \item Unilateral split: $G_S = \dfrac{1}{1-|S_{11}|^2}$, $G_0 = |S_{21}|^2$,
    $G_L = \dfrac{1}{1-|S_{22}|^2}$; dB $= 10\log G$.
  \item Given terminations (76 Bh: $Z_S = 25\,\Omega$, $Z_L = 40\,\Omega$):
    $\Gamma_S = \dfrac{Z_S - Z_0}{Z_S + Z_0}$, then
    $G_S = \dfrac{1-|\Gamma_S|^2}{|1-S_{11}\Gamma_S|^2}$, same form for $G_L$.
  \item $\Gamma_{in} = S_{11} + \dfrac{S_{12}S_{21}\Gamma_L}{1 - S_{22}\Gamma_L}$,
    $\Gamma_{out} = S_{22} + \dfrac{S_{12}S_{21}\Gamma_S}{1 - S_{11}\Gamma_S}$
    (74 Bh asks for both).
  \item Figure of merit $U = \dfrac{|S_{11}||S_{12}||S_{21}||S_{22}|}{(1-|S_{11}|^2)(1-|S_{22}|^2)}$,
    bound $\dfrac{1}{(1+U)^2} < \dfrac{G_T}{G_{TU}} < \dfrac{1}{(1-U)^2}$: the ``compare
    unilateral and bilateral'' sentence.
  \item Check: $G_{max} = \left|\dfrac{S_{21}}{S_{12}}\right|\left(K - \sqrt{K^2-1}\right)$;
    if $K < 1$ give $\mathrm{MSG} = |S_{21}|/|S_{12}|$.
  \item Circles (71 Bh prints $C$, $R$ directly): $\big||C| - R\big| > 1$ means the circle
    misses the chart, that port unconditionally stable; otherwise conditional, and the chart
    centre is stable when $|S_{11}| < 1$ (or $|S_{22}| < 1$).
\end{itemize}

\lead{Stub matching \yr{(Ch 2; double stub 17/24, single stub 6/24; only the chart is given)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item $\Gamma = \dfrac{Z_L - Z_0}{Z_L + Z_0}$, $z = Z/Z_0$, $y = 1/z$. Given $\Gamma$
    (82 Bh, 82 Ba, 81 Ba, 74 Bh): $z = \dfrac{1+\Gamma}{1-\Gamma}$. Given $Y$ (79 Bh):
    $y = YZ_0$.
  \item $\mathrm{VSWR} = \dfrac{1+|\Gamma|}{1-|\Gamma|}$,
    $Z_{in} = Z_0\dfrac{Z_L + jZ_0\tan\beta d}{Z_0 + jZ_L\tan\beta d}$, $\beta = 2\pi/\lambda$,
    $\lambda = c/f$ (physical lengths at 3 or 10 GHz).
  \item Stub admittance: short $y = -j\cot\beta\ell$, open $y = +j\tan\beta\ell$.
    $\lambda/2$ repeats the load; $\lambda/4$ gives $Z_0^2/Z_L$.
  \item Double-stub forbidden region $g > 1/\sin^2\beta d$; for $3\lambda/8$ spacing
    \mk{$g > 2$}.
  \item Network S-matrix (about 10 papers): before, $S_{11} = \Gamma_L$; after,
    $S_{11} = S_{22} = 0$, $S_{12} = S_{21} = e^{-j\theta}$.
\end{itemize}

\lead{S-matrix tests and devices \yr{(Ch 3 and Ch 4)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item Reciprocal $[S] = [S]^T$; lossless $[S]^\dagger[S] = [I]$ (unit columns, distinct
    columns orthogonal).
  \item $RL = -20\log|S_{11}|$, $IL = -20\log|S_{21}|$, $P_{ref} = |S_{11}|^2P_{in}$
    (79 Bh, 72 Ash).
  \item Magic tee $\dfrac{1}{\sqrt2}\begin{bmatrix}0&0&1&1\\0&0&1&-1\\1&1&0&0\\1&-1&0&0\end{bmatrix}$:
    E-arm column $+,-$; H-arm column $+,+$.
  \item Directional coupler: $C = 10\log P_1/P_4$, $D = 10\log P_4/P_3$,
    $I = 10\log P_1/P_3$, $\boxed{I = C + D}$.
\end{itemize}

\lead{Waveguides \yr{(Ch 4, 14/24 papers)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item $f_c = \dfrac{c}{2}\sqrt{(m/a)^2 + (n/b)^2}$; TE$_{10}$: $f_c = c/2a$, $\lambda_c = 2a$.
  \item $\lambda_g = \dfrac{\lambda}{\sqrt{1-(f_c/f)^2}}$,
    $\dfrac{1}{\lambda^2} = \dfrac{1}{\lambda_g^2} + \dfrac{1}{\lambda_c^2}$.
  \item $v_p = \dfrac{c}{\sqrt{1-(f_c/f)^2}}$, $v_g = c\sqrt{1-(f_c/f)^2}$, $v_pv_g = c^2$.
  \item $Z_{TE} = \dfrac{\eta}{\sqrt{1-(f_c/f)^2}}$, $Z_{TM} = \eta\sqrt{1-(f_c/f)^2}$,
    $\eta = 377\,\Omega$.
  \item TM$_{m0}$, TM$_{0n}$ do not exist, so TM$_{11}$ is the lowest TM. Circular:
    $f_c = \dfrac{p\,c}{2\pi a}$, $p'_{11} = 1.841$ (TE$_{11}$, dominant),
    $p_{01} = 2.405$ (TM$_{01}$).
\end{itemize}

\lead{Radiation hazards \yr{(Ch 7, 5/24 with numbers)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item Zones $R_1 = 0.62\sqrt{D^3/\lambda}$, $R_2 = 2D^2/\lambda$, \textbf{valid only for
    $D > \lambda$}; small antenna: $\lambda/2\pi$ and $2\lambda$.
  \item $S = \dfrac{PG}{4\pi R^2}$; $S = E^2/377$ in the far field only;
    $\mathrm{SAR} = \sigma E^2/\rho$.
  \item Limits: 4 W/kg whole body, 0.4 workers, 0.08 public; ICNIRP public $f/200$ W/m$^2$
    ($f$ in MHz, 400 MHz to 2 GHz).
\end{itemize}

\lead{Filters \yr{(Ch 6, 13/24, asked as design steps)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item Butterworth $P_{LR} = 1 + k^2(\omega/\omega_c)^{2N}$,
    $g_k = 2\sin\dfrac{(2k-1)\pi}{2N}$; $N$ from the attenuation requirement.
  \item Scale $L' = R_0L/\omega_c$, $C' = C/(R_0\omega_c)$; LPF to HPF
    $L' = R_0/(\omega_cC)$, $C' = 1/(R_0\omega_cL)$.
  \item Microstrip: short high-$Z$ line $=$ series $L$, short low-$Z$ line $=$ shunt $C$.
\end{itemize}

\lead{Measurements \yr{(Ch 8, theory questions; formulas worth quoting)}}
\begin{itemize}[leftmargin=12pt, itemsep=0.5pt, topsep=1pt]
  \item Double minimum, VSWR $> 10$: $S \approx \dfrac{\lambda_g}{\pi\Delta x}$.
    Static calorimeter $P = \dfrac{4.187\,mC_pT}{t}$.
  \item Bolometer double bridge $P = \dfrac{E_1^2 - E_2^2}{4R}$;
    $P_{av} = P_{peak}\times$ duty.
\end{itemize}
